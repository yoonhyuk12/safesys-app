function toUTF8(str: string): Uint8Array {
    return new TextEncoder().encode(str)
}

// ??? XML ?댁뒪耳?댄봽 ???
function esc(s: string | null | undefined): string {
    if (!s) return ''
    return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;')
}

// ??? ?대?吏 fetch ??base64 ???
async function fetchImageAsBuffer(url: string): Promise<{ data: Uint8Array; ext: string } | null> {
    try {
        const res = await fetch(url)
        const blob = await res.blob()
        const buf = await blob.arrayBuffer()
        const ext = blob.type.includes('png') ? 'png' : 'jpg'
        return { data: new Uint8Array(buf), ext }
    } catch {
        return null
    }
}

        emptyTbl2 += cell(0, 0, BODY_W, PHOTO_H, '', 0, 1, 1, 1, 2)
        emptyTbl2 += '</hp:tr>'
        body.push(table(emptyTbl2, 1, 1, BODY_W, PHOTO_H, 2))
    }

    // ?덈궡 臾멸뎄
    body.push(emptyPara())
    body.push(para('지적사항에 대한 조치 전, 조치 후 사진 필수', 5, 1))
    body.push(para('지적사항이 없는 경우 현장점검사진 첨부', 5, 1))

    // ?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧??
    // ?섎쾾?щ? (?덉쓣 ?뚮쭔)
    // ?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧?먥븧??
    const goodExContent = (inspection as any).good_example_content || ''
    if (goodExPhotos.length > 0 || goodExContent.trim()) {
        body.push(para('<수범사례 사진대지>', 6, 1, true))
        body.push(emptyPara())

        const geInfoColW = [Math.round(BODY_W * 0.15), 0]
        geInfoColW[1] = BODY_W - geInfoColW[0]

        let geInfoRows = '<hp:tr>'
        geInfoRows += cell(0, 0, geInfoColW[0], INFO_ROW_H, '지구명', 3, 1, 1, 1, 3)
        geInfoRows += cell(1, 0, geInfoColW[1], INFO_ROW_H, inspection.district_name || '', 0, 1, 1, 1, 2)
        geInfoRows += '</hp:tr><hp:tr>'
        geInfoRows += cell(0, 1, geInfoColW[0], INFO_ROW_H, '설 명', 3, 1, 1, 1, 3)
        geInfoRows += cell(1, 1, geInfoColW[1], INFO_ROW_H, goodExContent, 0, 0, 1, 1, 2, 'TOP')
        geInfoRows += '</hp:tr>'
        body.push(table(geInfoRows, 2, 2, BODY_W, INFO_ROW_H * 2))

        const GE_PHOTO_H = Math.round(70 * MM)
        if (goodExImgId) {
            let pRow = '<hp:tr>'
            pRow += imageCell(0, 0, BODY_W, GE_PHOTO_H, goodExImgId, Math.round(55 * MM), Math.round(40 * MM), 2)
            pRow += '</hp:tr>'
            body.push(table(pRow, 1, 1, BODY_W, GE_PHOTO_H, 2))
        } else {
            let pRow = '<hp:tr>'
            pRow += cell(0, 0, BODY_W, GE_PHOTO_H, '', 0, 1, 1, 1, 2)
            pRow += '</hp:tr>'
            body.push(table(pRow, 1, 1, BODY_W, GE_PHOTO_H, 2))
        }
    }

    // section0.xml 議곕┰
    const sectionXml = `<?xml version="1.0" encoding="UTF-8" standalone="yes" ?>
<hs:sec xmlns:ha="http://www.hancom.co.kr/hwpml/2011/app" xmlns:hp="http://www.hancom.co.kr/hwpml/2011/paragraph" xmlns:hp10="http://www.hancom.co.kr/hwpml/2016/paragraph" xmlns:hs="http://www.hancom.co.kr/hwpml/2011/section" xmlns:hc="http://www.hancom.co.kr/hwpml/2011/core" xmlns:hh="http://www.hancom.co.kr/hwpml/2011/head" xmlns:hhs="http://www.hancom.co.kr/hwpml/2011/history" xmlns:hm="http://www.hancom.co.kr/hwpml/2011/master-page" xmlns:hpf="http://www.hancom.co.kr/schema/2011/hpf" xmlns:dc="http://purl.org/dc/elements/1.1/" xmlns:opf="http://www.idpf.org/2007/opf/" xmlns:ooxmlchart="http://www.hancom.co.kr/hwpml/2016/ooxmlchart" xmlns:hwpunitchar="http://www.hancom.co.kr/hwpml/2016/HwpUnitChar" xmlns:epub="http://www.idpf.org/2007/ops" xmlns:config="urn:oasis:names:tc:opendocument:xmlns:config:1.0">
${body.join('\n')}
</hs:sec>`

    // ???????????????????????????????????
    // ZIP 議곕┰ (?ㅼ젣 ?쒓??ㅽ뵾??異쒕젰 援ъ“??留욎땄)
    // 紐⑤뱺 XML? TextEncoder濡?UTF-8 諛붿씠?몃줈 蹂?섑븯?????
    // ???????????????????????????????????
    // mimetype (한글 구버전 호환을 위해 vnd 타입 사용)
    zip.file('mimetype', toUTF8('application/vnd.hancom.hwpx'), { compression: 'STORE' })

    // version.xml (호환성이 높은 구버전 포맷)
    zip.file('version.xml', toUTF8(`<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n<version><releaseVersion major="1" minor="4" build="0" revision="0"/><appVersion text="HWP 2.5.3.0" date="2016-01-19" format="1.4" compatibleVersion="1.4"/></version>`), { compression: 'STORE' })

    // content.hpf 기준 상대경로 호환: Contents/BinData/*
    for (const img of images) {
        zip.file(`Contents/BinData/${img.filename}`, img.data)
    }

    // Contents/header.xml
    zip.file('Contents/header.xml', toUTF8(headerXml))

    // Contents/section0.xml
    zip.file('Contents/section0.xml', toUTF8(sectionXml))

    // Preview
    zip.file('Preview/PrvText.txt', toUTF8(''))

    // settings.xml (최소 설정)
    zip.file('settings.xml', toUTF8(`<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n<config:config-item-set xmlns:config="urn:oasis:names:tc:opendocument:xmlns:config:1.0" config:name="HWP-Settings"/>`))

    // Contents/content.hpf (manifest)
    const imgManifest = images.map(img =>
        `<item id="${img.id}" href="BinData/${img.filename}" media-type="image/${img.ext === 'jpg' ? 'jpeg' : img.ext}" isEmbeded="1"/>`
    ).join('')

    zip.file('Contents/content.hpf', toUTF8(`<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n<package xmlns="http://www.idpf.org/2007/opf/" version="2.0">\n<metadata>\n<dc:title xmlns:dc="http://purl.org/dc/elements/1.1/">안전점검 보고서</dc:title>\n<dc:format xmlns:dc="http://purl.org/dc/elements/1.1/">application/vnd.hancom.hwpx</dc:format>\n<dc:language xmlns:dc="http://purl.org/dc/elements/1.1/">ko-KR</dc:language>\n</metadata>\n<manifest>\n<item id="header" href="header.xml" media-type="application/vnd.hancom.hwpml+xml"/>\n<item id="section0" href="section0.xml" media-type="application/vnd.hancom.hwpml+xml"/>\n${imgManifest}\n</manifest>\n<spine><itemref idref="section0"/></spine>\n</package>`))

    // META-INF/container.xml
    zip.file('META-INF/container.xml', toUTF8(`<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n<container version="1.0" xmlns="urn:oasis:names:tc:opendocument:xmlns:container"><rootfiles><rootfile full-path="Contents/content.hpf" media-type="application/vnd.hancom.hpf+xml"/></rootfiles></container>`))

    // META-INF/manifest.xml
    zip.file('META-INF/manifest.xml', toUTF8(`<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n<manifest xmlns="urn:oasis:names:tc:opendocument:xmlns:manifest:1.0"><file-entry full-path="/" media-type="application/vnd.hancom.hwpx"/><file-entry full-path="/Contents/" media-type="application/vnd.hancom.hwpx"/></manifest>`))

    // ?뚯씪 ?앹꽦 諛??ㅼ슫濡쒕뱶
    const blob = await zip.generateAsync({
        type: 'blob',
        compression: 'DEFLATE',
        compressionOptions: { level: 6 }
    })
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = `safety_inspection_${inspection.inspection_type}_${inspection.inspection_date}.hwpx`
    a.click()
    URL.revokeObjectURL(url)
}
